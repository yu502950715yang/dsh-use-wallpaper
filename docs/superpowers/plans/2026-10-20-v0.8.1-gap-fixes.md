function collectSounds(root: any): SceneSoundObject[] | undefined {
  const out: SceneSoundObject[] = [];
  for (const o of Array.isArray(root.objects) ? root.objects : []) {
    if (!Array.isArray(o?.sound)) continue;
    const files = o.sound.filter((s: unknown): s is string => typeof s === 'string' && !!s);
    if (files.length === 0) continue;
    const volume = typeof o.volume === 'number' && Number.isFinite(o.volume) ? Math.max(0, Math.min(1, o.volume)) : 1;
    const mode = o.playbackmode === 'random' || o.playbackmode === 'single' ? o.playbackmode : 'loop';
    out.push({ files, volume, playbackmode: mode });
  }
  return out.length > 0 ? out : undefined;
}
```

    /// 预推进 starttime 秒（WE 语义：起播时系统已运行 N 秒）。固定 1/60 步长、上限 900 步。
    pub fn advance_initial(&mut self, seconds: f32) {
        if !(seconds > 0.0) {
            return;
        }
        let steps = ((seconds * 60.0).ceil() as u32).min(900);
        for _ in 0..steps {
            self.update(1.0 / 60.0);
        }
    }

// WE 的 {user,value} / {script,value} 包装：wasm 侧只认标量 → 递归解成 value。
function normalizeOverrideValue(v: unknown): unknown {
  if (Array.isArray(v)) return v.map(normalizeOverrideValue);
  if (v && typeof v === 'object') {
    const o = v as Record<string, unknown>;
    if ('value' in o && (('user' in o) || ('script' in o))) return normalizeOverrideValue(o.value);
    const out: Record<string, unknown> = {};
    for (const [k, x] of Object.entries(o)) out[k] = normalizeOverrideValue(x);
    return out;
  }
  return v;
}
```
