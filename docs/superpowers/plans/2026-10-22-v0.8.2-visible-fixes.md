```ts
import type { UtilLayerKind } from '../shared/types.js';

export interface UtilMaterialSpec {
  layerKind: UtilLayerKind;
  kind: 'passthrough' | 'solid';
  color: [number, number, number];
  opacity: number;
}

// util 层内置材质：solid 纯色；其余直通（真实效果由效果链/合成叠加产生）
export function resolveUtilMaterial(
  layerKind: UtilLayerKind,
  overrides: { color?: [number, number, number]; alpha?: number } = {},
): UtilMaterialSpec {
  const color = overrides.color ?? [255, 255, 255];
  const opacity = typeof overrides.alpha === 'number' && Number.isFinite(overrides.alpha) ? Math.min(Math.max(overrides.alpha, 0), 1) : 1;
  return layerKind === 'solid'
    ? { layerKind, kind: 'solid', color, opacity }
    : { layerKind, kind: 'passthrough', color, opacity };
}
```

`object-range.ts`：

```ts
import type { UtilLayerKind } from '../shared/types.js';

// util 层世界尺寸：project/fullscreen 覆盖整个正交视口；compose/solid 按 size×|scale|
export function utilWorldSize(
  kind: UtilLayerKind,
  size: [number, number] | undefined,
  scale: [number, number],
  sceneW: number,
  sceneH: number,
): { w: number; h: number } {
  if (kind === 'project' || kind === 'fullscreen') return { w: Math.abs(sceneW), h: Math.abs(sceneH) };
  const s = size ?? [1, 1];
  return { w: Math.abs(s[0] * scale[0]), h: Math.abs(s[1] * scale[1]) };
}
```
